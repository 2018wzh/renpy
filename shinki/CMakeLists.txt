cmake_minimum_required(VERSION 3.25)
project(shinki-emu-renpy LANGUAGES C CXX)
if(CMAKE_SIZEOF_VOID_P EQUAL 0)
    message(FATAL_ERROR "Ren'Py adapter requires a 64-bit target")
endif()
set(RENPY_SDK_ROOT "" CACHE PATH "Official Ren'Py SDK root")
set(RENPY_PYTHON_DEV "" CACHE PATH "CPython development root")
set(RENPY_SDL_INCLUDE "" CACHE PATH "SDL include directory")
set(RENPY_IMPORT_LIBRARY_DIR "" CACHE PATH "Generated import libraries")
set(RENPY_CYTHON_LICENSE "" CACHE FILEPATH "Cython license")
set(RENPY_FFMPEG_ROOT "$ENV{SHINKI_EMU_FFMPEG_INSTALLED}" CACHE PATH "Shared FFmpeg installation")
option(RENPY_STATIC "Embed the adapter and its Python display module" OFF)
set(RENPY_STATIC_ROOT "" CACHE PATH "Target CPython headers and static runtime libraries")
get_filename_component(RENPY_CYTHON_LICENSE_DIR "${RENPY_CYTHON_LICENSE}" DIRECTORY)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
get_filename_component(RENPY_SOURCE "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
if(WIN32)
    set(RENPY_PYTHON_INCLUDE "${RENPY_PYTHON_DEV}/include")
    set(RENPY_DISPLAY_SUFFIX ".pyd")
    set(RENPY_NATIVE "${RENPY_SDK_ROOT}/lib/py3-windows-x86_64")
    set(RENPY_SUFFIX ".dll")
elseif(APPLE)
    set(RENPY_PYTHON_INCLUDE "${RENPY_PYTHON_DEV}/Include" "${RENPY_PYTHON_DEV}")
    set(RENPY_DISPLAY_SUFFIX ".so")
    set(RENPY_NATIVE "${RENPY_SDK_ROOT}/lib/py3-mac-universal")
    set(RENPY_SUFFIX ".dylib")
else()
    set(RENPY_PYTHON_INCLUDE "${RENPY_PYTHON_DEV}/Include" "${RENPY_PYTHON_DEV}")
    set(RENPY_DISPLAY_SUFFIX ".so")
    set(RENPY_NATIVE "${RENPY_SDK_ROOT}/lib/py3-linux-x86_64")
    set(RENPY_SUFFIX ".so")
endif()
if(RENPY_STATIC)
    set(RENPY_PYTHON_INCLUDE "${RENPY_STATIC_ROOT}/include/python3.12")
    set(RENPY_SDL_INCLUDE "${RENPY_STATIC_ROOT}/include/SDL2")
endif()
list(GET RENPY_PYTHON_INCLUDE 0 RENPY_PYTHON_HEADERS)
foreach(required "${RENPY_PYTHON_HEADERS}/Python.h" "${RENPY_SDL_INCLUDE}/SDL.h" "${RENPY_CYTHON_LICENSE}")
    if(NOT EXISTS "${required}")
        message(FATAL_ERROR "Missing Ren'Py SDK development input: ${required}")
    endif()
endforeach()
file(COPY "${RENPY_SDL_INCLUDE}/" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/headers/SDL2")
file(GLOB_RECURSE cython_dependencies CONFIGURE_DEPENDS "${RENPY_SOURCE}/renpy/*.pxd" "${RENPY_SOURCE}/src/*.pxd")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/display.c"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/generate_display.py"
        "${RENPY_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/display.c" "${RENPY_STATIC}"
    DEPENDS "${RENPY_SOURCE}/renpy/pygame/display.pyx" "${CMAKE_CURRENT_LIST_DIR}/generate_display.py" ${cython_dependencies} VERBATIM)
if(RENPY_STATIC)
    add_library(shinki_renpy_display OBJECT "${CMAKE_CURRENT_BINARY_DIR}/display.c")
else()
    add_library(shinki_renpy_display MODULE "${CMAKE_CURRENT_BINARY_DIR}/display.c")
endif()
target_include_directories(shinki_renpy_display PRIVATE ${RENPY_PYTHON_INCLUDE} "${CMAKE_CURRENT_BINARY_DIR}/headers" "${RENPY_SOURCE}/src" "${RENPY_SOURCE}/src/pygame/include")
if(RENPY_STATIC AND EXISTS "${RENPY_STATIC_ROOT}/include/shinki-renpy-symbols.h")
    target_compile_options(shinki_renpy_display PRIVATE
        "-include" "${RENPY_STATIC_ROOT}/include/shinki-renpy-symbols.h")
endif()
if(RENPY_STATIC)
    target_compile_definitions(shinki_renpy_display PRIVATE RENPY_STATIC=1)
elseif(WIN32)
    target_link_libraries(shinki_renpy_display PRIVATE "${RENPY_IMPORT_LIBRARY_DIR}/libpython3.12.lib" "${RENPY_IMPORT_LIBRARY_DIR}/librenpython.lib")
    target_link_options(shinki_renpy_display PRIVATE /NODEFAULTLIB:python312.lib)
    target_compile_options(shinki_renpy_display PRIVATE /U_DEBUG)
elseif(APPLE)
    target_link_libraries(shinki_renpy_display PRIVATE "${RENPY_NATIVE}/librenpython.dylib")
    set_target_properties(shinki_renpy_display PROPERTIES INSTALL_RPATH "@loader_path")
else()
    target_link_directories(shinki_renpy_display PRIVATE "${RENPY_NATIVE}")
    target_link_libraries(shinki_renpy_display PRIVATE renpython dl)
    set_target_properties(shinki_renpy_display PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
set_target_properties(shinki_renpy_display PROPERTIES PREFIX "" OUTPUT_NAME display SUFFIX "${RENPY_DISPLAY_SUFFIX}" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/overrides" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/overrides")
if(RENPY_STATIC)
    add_library(shinki_emu_renpy STATIC core.cpp $<TARGET_OBJECTS:shinki_renpy_display>)
    target_compile_definitions(shinki_emu_renpy PRIVATE RENPY_STATIC=1)
else()
    add_library(shinki_emu_renpy SHARED core.cpp)
endif()
target_compile_features(shinki_emu_renpy PRIVATE cxx_std_17)
target_include_directories(shinki_emu_renpy PRIVATE ${RENPY_PYTHON_INCLUDE} "${CMAKE_CURRENT_BINARY_DIR}/headers" "${RENPY_SDL_INCLUDE}" "${CMAKE_CURRENT_LIST_DIR}/vendor/shinki-emu-abi/include")
if(RENPY_STATIC AND EXISTS "${RENPY_STATIC_ROOT}/include/shinki-renpy-symbols.h")
    target_compile_options(shinki_emu_renpy PRIVATE
        "-include" "${RENPY_STATIC_ROOT}/include/shinki-renpy-symbols.h")
endif()
if(RENPY_STATIC)
    # Cargo emits the complete target archive closure after the adapter archive.
elseif(WIN32)
    target_link_libraries(shinki_emu_renpy PRIVATE "${RENPY_IMPORT_LIBRARY_DIR}/libpython3.12.lib" "${RENPY_IMPORT_LIBRARY_DIR}/librenpython.lib")
    target_link_options(shinki_emu_renpy PRIVATE /NODEFAULTLIB:python312.lib)
    target_compile_options(shinki_emu_renpy PRIVATE /U_DEBUG)
elseif(APPLE)
    target_link_libraries(shinki_emu_renpy PRIVATE "${RENPY_NATIVE}/librenpython.dylib")
    set_target_properties(shinki_emu_renpy PROPERTIES INSTALL_RPATH "@loader_path/Resources")
else()
    target_link_directories(shinki_emu_renpy PRIVATE "${RENPY_NATIVE}")
    target_link_libraries(shinki_emu_renpy PRIVATE renpython dl)
    set_target_properties(shinki_emu_renpy PROPERTIES INSTALL_RPATH "$ORIGIN/Resources")
endif()
if(RENPY_FFMPEG_ROOT)
    target_link_directories(shinki_emu_renpy PRIVATE "${RENPY_FFMPEG_ROOT}/lib")
    target_link_libraries(shinki_emu_renpy PRIVATE avcodec avformat avutil swresample swscale)
endif()
set_target_properties(shinki_emu_renpy PROPERTIES OUTPUT_NAME shinki_emu_renpy)
if(NOT RENPY_STATIC)
    set_target_properties(shinki_emu_renpy PROPERTIES PREFIX "")
endif()
add_dependencies(shinki_emu_renpy shinki_renpy_display)
install(TARGETS shinki_emu_renpy RUNTIME DESTINATION . LIBRARY DESTINATION . ARCHIVE DESTINATION lib)
if(RENPY_STATIC)
    # The display override is a builtin module in the adapter archive.
elseif(WIN32)
    install(TARGETS shinki_renpy_display LIBRARY DESTINATION Resources/overrides RUNTIME DESTINATION Resources/overrides)
else()
    install(TARGETS shinki_renpy_display LIBRARY DESTINATION Resources)
endif()
if(RENPY_STATIC)
    # Native runtime code is linked into Manager, never loaded from resources.
elseif(WIN32)
    foreach(dll libpython3.12.dll librenpython.dll libwinpthread-1.dll libEGL.dll libGLESv2.dll)
        install(FILES "${RENPY_NATIVE}/${dll}" DESTINATION .)
    endforeach()
elseif(APPLE)
    install(FILES "${RENPY_NATIVE}/librenpython.dylib" DESTINATION Resources)
    install(CODE "
        execute_process(COMMAND install_name_tool -id @rpath/librenpython.dylib
            \"\${CMAKE_INSTALL_PREFIX}/Resources/librenpython.dylib\"
            COMMAND_ERROR_IS_FATAL ANY)
        foreach(binary shinki_emu_renpy.dylib Resources/display.so)
            execute_process(COMMAND install_name_tool
                -change @executable_path/librenpython.dylib @rpath/librenpython.dylib
                \"\${CMAKE_INSTALL_PREFIX}/\${binary}\"
                COMMAND_ERROR_IS_FATAL ANY)
        endforeach()
    ")
else()
    foreach(lib librenpython.so)
        install(FILES "${RENPY_NATIVE}/${lib}" DESTINATION Resources)
    endforeach()
endif()
if(RENPY_STATIC)
    install(DIRECTORY "${RENPY_STATIC_ROOT}/Resources/" DESTINATION Resources)
else()
    install(DIRECTORY "${RENPY_SDK_ROOT}/lib/python3.12" DESTINATION Resources/lib PATTERN "__pycache__" EXCLUDE)
endif()
install(DIRECTORY "${RENPY_SOURCE}/renpy/" DESTINATION Resources/renpy PATTERN "*.pyx" EXCLUDE PATTERN "*.pxd" EXCLUDE PATTERN "*.pxi" EXCLUDE PATTERN "*.pyi" EXCLUDE PATTERN "__pycache__" EXCLUDE PATTERN ".git*" EXCLUDE)
install(FILES "${RENPY_SOURCE}/renpy.py" "${CMAKE_CURRENT_LIST_DIR}/shinki_renpy_bridge.py" "${CMAKE_CURRENT_LIST_DIR}/shinki_save_migration.py" DESTINATION Resources)
install(FILES "${RENPY_SDK_ROOT}/renpy/vc_version.py" DESTINATION Resources/renpy)
install(FILES "${CMAKE_CURRENT_LIST_DIR}/sdk.json" DESTINATION Resources)
