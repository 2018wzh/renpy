cmake_minimum_required(VERSION 3.25)
project(shinki-emu-renpy LANGUAGES C CXX)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The current Ren'Py SDK adapter requires MSVC Windows x64")
endif()
set(RENPY_SDK_ROOT "" CACHE PATH "Official Ren'Py 8.5.3 SDK root")
set(RENPY_PYTHON_DEV "" CACHE PATH "CPython 3.12.7 development root (containing include)")
set(RENPY_SDL_INCLUDE "" CACHE PATH "SDL 2.0.20 include directory")
set(RENPY_IMPORT_LIBRARY_DIR "" CACHE PATH "MSVC import libraries generated by shinki-emu-build")
set(RENPY_CYTHON_LICENSE "" CACHE FILEPATH "License text from official Cython 3.2.4")
get_filename_component(RENPY_CYTHON_LICENSE_DIR "${RENPY_CYTHON_LICENSE}" DIRECTORY)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
get_filename_component(RENPY_SOURCE "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(RENPY_NATIVE "${RENPY_SDK_ROOT}/lib/py3-windows-x86_64")
foreach(required "${RENPY_NATIVE}/librenpython.dll" "${RENPY_NATIVE}/libpython3.12.dll"
        "${RENPY_PYTHON_DEV}/include/Python.h" "${RENPY_PYTHON_DEV}/LICENSE.txt"
        "${RENPY_SDL_INCLUDE}/SDL.h" "${RENPY_CYTHON_LICENSE}" "${RENPY_CYTHON_LICENSE_DIR}/COPYING.txt"
        "${RENPY_IMPORT_LIBRARY_DIR}/libpython3.12.lib" "${RENPY_IMPORT_LIBRARY_DIR}/librenpython.lib")
    if(NOT EXISTS "${required}")
        message(FATAL_ERROR "Missing pinned Ren'Py SDK development input: ${required}")
    endif()
endforeach()
file(COPY "${RENPY_SDL_INCLUDE}/" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/headers/SDL2")
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import Cython; assert Cython.__version__ == '3.2.4'"
    RESULT_VARIABLE cython_status)
if(NOT cython_status EQUAL 0)
    message(FATAL_ERROR "The build interpreter must provide Cython 3.2.4")
endif()
file(READ "${RENPY_SDL_INCLUDE}/SDL_version.h" sdl_version)
foreach(version_define "#define SDL_MAJOR_VERSION[ \t]+2" "#define SDL_MINOR_VERSION[ \t]+0"
        "#define SDL_PATCHLEVEL[ \t]+20")
    if(NOT sdl_version MATCHES "${version_define}([\r\n]|$)")
        message(FATAL_ERROR "RENPY_SDL_INCLUDE must provide SDL 2.0.20 headers")
    endif()
endforeach()

file(GLOB_RECURSE cython_dependencies CONFIGURE_DEPENDS "${RENPY_SOURCE}/renpy/*.pxd" "${RENPY_SOURCE}/src/*.pxd")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/display.c"
    COMMAND "${Python3_EXECUTABLE}" -m cython -3 -X profile=False -X embedsignature=True
        -X embedsignature.format=python -I "${RENPY_SOURCE}" -I "${RENPY_SOURCE}/src"
        -I "${RENPY_SOURCE}/src/pygame/include" "${RENPY_SOURCE}/renpy/pygame/display.pyx"
        -o "${CMAKE_CURRENT_BINARY_DIR}/display.c"
    DEPENDS "${RENPY_SOURCE}/renpy/pygame/display.pyx" ${cython_dependencies}
    VERBATIM)
add_library(shinki_renpy_display MODULE "${CMAKE_CURRENT_BINARY_DIR}/display.c")
target_include_directories(shinki_renpy_display PRIVATE "${RENPY_PYTHON_DEV}/include"
    "${CMAKE_CURRENT_BINARY_DIR}/headers" "${RENPY_SOURCE}/src" "${RENPY_SOURCE}/src/pygame/include")
target_link_libraries(shinki_renpy_display PRIVATE
    "${RENPY_IMPORT_LIBRARY_DIR}/libpython3.12.lib"
    "${RENPY_IMPORT_LIBRARY_DIR}/librenpython.lib")
# MSVC's Python headers name the stock interpreter import library. The SDK
# uses its own MinGW-built libpython3.12.dll; never bind a second interpreter.
target_link_options(shinki_renpy_display PRIVATE /NODEFAULTLIB:python312.lib)
target_compile_options(shinki_renpy_display PRIVATE /U_DEBUG)
set_target_properties(shinki_renpy_display PROPERTIES PREFIX "" OUTPUT_NAME display SUFFIX ".pyd"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/overrides"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/overrides")

add_library(shinki_emu_renpy SHARED core.cpp)
target_compile_features(shinki_emu_renpy PRIVATE cxx_std_17)
target_compile_definitions(shinki_emu_renpy PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(shinki_emu_renpy PRIVATE "${RENPY_PYTHON_DEV}/include"
    "${CMAKE_CURRENT_BINARY_DIR}/headers" "${RENPY_SDL_INCLUDE}"
    "${CMAKE_CURRENT_LIST_DIR}/vendor/shinki-emu-abi/include")
target_link_libraries(shinki_emu_renpy PRIVATE
    "${RENPY_IMPORT_LIBRARY_DIR}/libpython3.12.lib"
    "${RENPY_IMPORT_LIBRARY_DIR}/librenpython.lib")
target_link_options(shinki_emu_renpy PRIVATE /NODEFAULTLIB:python312.lib)
target_compile_options(shinki_emu_renpy PRIVATE /U_DEBUG)
set_target_properties(shinki_emu_renpy PROPERTIES PREFIX "" OUTPUT_NAME shinki_emu_renpy)
add_dependencies(shinki_emu_renpy shinki_renpy_display)

install(TARGETS shinki_emu_renpy RUNTIME DESTINATION . ARCHIVE DESTINATION lib)
install(TARGETS shinki_renpy_display LIBRARY DESTINATION Resources/overrides
    RUNTIME DESTINATION Resources/overrides)
foreach(dll libpython3.12.dll librenpython.dll libwinpthread-1.dll libEGL.dll libGLESv2.dll)
    install(FILES "${RENPY_NATIVE}/${dll}" DESTINATION .)
endforeach()
# Ship only the open-source runtime, not SDK launchers, Steam/Cubism extensions,
# nvdrs driver-profile modifications, or Microsoft's redistributable shader DLL.
# On Windows ANGLE may use the operating system's D3DCompiler_47.dll.
install(DIRECTORY "${RENPY_SDK_ROOT}/lib/python3.12" DESTINATION Resources/lib
    PATTERN "__pycache__" EXCLUDE)
install(DIRECTORY "${RENPY_SOURCE}/renpy/" DESTINATION Resources/renpy
    PATTERN "*.pyx" EXCLUDE PATTERN "*.pxd" EXCLUDE PATTERN "*.pxi" EXCLUDE
    PATTERN "*.pyi" EXCLUDE PATTERN "__pycache__" EXCLUDE PATTERN ".git*" EXCLUDE)
install(FILES "${RENPY_SOURCE}/renpy.py" "${CMAKE_CURRENT_LIST_DIR}/shinki_renpy_bridge.py"
    DESTINATION Resources)
# Generated by the SDK release build, deliberately not checked into upstream.
install(FILES "${RENPY_SDK_ROOT}/renpy/vc_version.py" DESTINATION Resources/renpy)
install(FILES "${RENPY_SDK_ROOT}/LICENSE.txt"
    DESTINATION licenses/renpy RENAME LICENSE-RenPy.txt)
install(FILES "${RENPY_PYTHON_DEV}/LICENSE.txt"
    DESTINATION licenses/renpy RENAME LICENSE-CPython.txt)
install(FILES "${RENPY_CYTHON_LICENSE}"
    DESTINATION licenses/renpy RENAME LICENSE-Cython.txt)
install(FILES "${RENPY_CYTHON_LICENSE_DIR}/COPYING.txt"
    DESTINATION licenses/renpy RENAME COPYING-Cython.txt)
install(FILES "${RENPY_NATIVE}/libwinpthread-1.txt" DESTINATION licenses/renpy)
install(FILES "${CMAKE_CURRENT_LIST_DIR}/LICENSE"
    DESTINATION licenses/renpy RENAME LICENSE-SHINKI.txt)
install(FILES "${CMAKE_CURRENT_LIST_DIR}/sdk.json" DESTINATION Resources)
